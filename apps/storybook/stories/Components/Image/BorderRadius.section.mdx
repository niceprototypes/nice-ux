import { StoryHeader } from "@niceprototypes/docs/components/Story"
import Image from "@niceprototypes/image"
import StoryDescription from "../../../src/components/StoryDescription"
import { DEMO_CODE_COLUMNS } from "@niceprototypes/docs/components/DataTable"
import { DataTable } from "@niceprototypes/docs/components/DataTable"
import { toSource } from "@niceprototypes/docs/services"

export const label = "borderRadius"

<StoryHeader title="borderRadius" />

<StoryDescription property="border-radius">
  Rounds the image's corners using borderRadius tokens.
</StoryDescription>

<DataTable
  showHeader={false}
  columns={DEMO_CODE_COLUMNS}
  tables={[{ rows: ["smaller", "small", "base", "large", "larger"].map(radius => {
    const props = {
      vendor: true,
      alt: `${radius} radius demo`,
      width: "240px",
      height: "320px",
      backgroundSize: "cover",
      borderRadius: radius,
    }
    return {
      id: String(radius),
      values: [
        <Image {...props} />,
        toSource("Image", props),
      ],
    }
  }) }]}
/>
