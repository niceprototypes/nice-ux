  <code>margin</code>.
</Typography>

<div>
  {["padding", "margin"].map(value => (
    <DocsRow key={value}>
      <Flex direction="column">

          style={{

          }}
        >
          <Flex

              borderRadius: `calc(${getToken("cellHeight", "smaller").var} / 2 + 2px)`,
            }}
          >
            <DemoDots dots={twoDots} />
          </Flex>
        </Flex>
      </Flex>
      <Source
        language="tsx"
        code={`<Flex mode="${value}">\n  {children}\n</Flex>`}
      />
    </DocsRow>
