import { StoryHeader } from "@niceprototypes/docs/components/Story"
import Icon from "@niceprototypes/icon"
import StoryDescription from "../../../src/components/StoryDescription"
import { DEMO_CODE_COLUMNS } from "@niceprototypes/docs/components/DataTable"
import { DataTable } from "@niceprototypes/docs/components/DataTable"
import { animationDurations } from "./Icon.services"

export const label = "animationDuration"

<StoryHeader title="animationDuration" />

<StoryDescription
  kind="css"
  property="animation-duration"
>
  Sets the speed of the icon's animation. Icon durations run one step slower
  than base element durations (fast 240ms / base 480ms / slow 960ms).
</StoryDescription>

<DataTable
  showHeader={false}
  columns={DEMO_CODE_COLUMNS}
  tables={[{ rows: animationDurations.map(duration => ({
    id: String(duration),
    values: [
      <Icon
        animation="spin"
        animationDuration={duration}
        name="spinner"
        size="larger"
      />,
      duration === "base"
        ? `<Icon name="spinner" animation="spin" />`
        : `<Icon name="spinner" animation="spin" animationDuration="${duration}" />`,
    ],
  })) }]}
/>
