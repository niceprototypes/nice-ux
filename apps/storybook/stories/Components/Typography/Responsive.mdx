import { Meta, Source } from '@storybook/addon-docs/blocks'
import Typography from '../../../src/nice/Typography'
import { getToken, getSizeToken, getBreakpoint } from 'nice-react-styles'
import { DocsRow } from '../../../src/components/DocsRow'
import { toSource } from '../../../src/services'

export const mobileLabel = `≤${getBreakpoint("mobile").value}px`
export const desktopLabel = `≥${getBreakpoint("desktop").value}px`

<Meta title="React/Components/Typography/responsive" />

# Typography — Responsive

<Typography color="light">Typography sizes scale automatically across breakpoints — the same <code>size</code> token resolves to different values on mobile and desktop.</Typography>

<div>
  {["smaller", "small", "base", "large", "larger"].map(size => {
    const mobile = getToken("fontSize", size).value
    const desktop = getSizeToken("fontSize", size, "desktop").value
    const children = `${size} — ${mobileLabel}: ${mobile}, ${desktopLabel}: ${desktop}`
    return (
      <DocsRow key={size}>
        <Typography size={size}>{children}</Typography>
        <Source language="tsx" code={toSource("Typography", { size }, { children })} />
      </DocsRow>
    )
  })}
</div>
