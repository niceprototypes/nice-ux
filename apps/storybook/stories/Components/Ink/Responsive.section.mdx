import { StoryHeader } from "@niceprototypes/docs/components/Story"
import Ink from "@niceprototypes/docs/nice/Ink"
import { getToken, getBreakpointValue } from "@niceprototypes/styles"
import { DEMO_CODE_COLUMNS } from "@niceprototypes/docs/components/DataTable"
import { DataTable } from "@niceprototypes/docs/components/DataTable"
import { toSource } from "@niceprototypes/docs/services"

export const smallLabel = `≤${getBreakpointValue("phone")}px`
export const largeLabel = `≥${getBreakpointValue("laptop")}px`

export const label = "responsive"

<StoryHeader title="responsive" />

<Ink><Ink color="light">Ink sizes scale automatically across breakpoints — the same</Ink> <Ink color="light" code>size</Ink> <Ink color="light">token resolves to different values at the small and large breakpoints.</Ink></Ink>

<DataTable
  showHeader={false}
  columns={DEMO_CODE_COLUMNS}
  tables={[{ rows: ["smaller", "small", "base", "large", "larger"].map(size => {
    const smallValue = getToken("fontSize", size, { as: "value" })
    const largeValue = getToken("fontSize", size, { breakpoint: "laptop", as: "value" })
    const children = `${size} — ${smallLabel}: ${smallValue}, ${largeLabel}: ${largeValue}`
    return {
        id: String(size),
        values: [
          <Ink size={size}>{children}</Ink>,
          toSource("Ink", { size }, { children }),
        ],
    }
  }) }]}
/>
