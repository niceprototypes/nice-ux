import { StoryHeader } from "@niceprototypes/docs/components/Story"
import Ink from "@niceprototypes/docs/nice/Ink"
import StoryDescription from "../../../src/components/StoryDescription"
import { DEMO_CODE_COLUMNS } from "@niceprototypes/docs/components/DataTable"
import { DataTable } from "@niceprototypes/docs/components/DataTable"
import { toSource } from "@niceprototypes/docs/services"

export const loremIpsumLong =
  "Lorem ipsum dolor sit amet, consectetur adipiscing elit. Sed do eiusmod tempor incididunt ut labore et dolore magna aliqua. Ut enim ad minim veniam, quis nostrud exercitation ullamco laboris nisi ut aliquip ex ea commodo consequat."

export const label = "lineHeight"

<StoryHeader title="lineHeight" />

<StoryDescription property="line-height">
  Sets the vertical spacing between lines of text using lineHeight tokens.
</StoryDescription>

<DataTable
  showHeader={false}
  columns={DEMO_CODE_COLUMNS}
  tables={[{ rows: ["condensed", "base", "expanded"].map(lineHeight => ({
    id: String(lineHeight),
    values: [
          <Ink lineHeight={lineHeight}>{loremIpsumLong}</Ink>,
          toSource(
          "Ink",
          { lineHeight },
          { children: loremIpsumLong }
        ),
        ],
  })) }]}
/>
