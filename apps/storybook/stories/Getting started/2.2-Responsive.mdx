# Styles + responsive

```bash
npm install nice-react-styles styled-components
```

```tsx
import styled from "styled-components"
import { getToken } from "nice-react-styles"

const Heading = styled.h1`
  font-size: ${getToken("fontSize", "larger")};
`
```

```tsx
import styled from "styled-components"
import { getBreakpoint, getToken } from "nice-react-styles"

const Wrapper = styled.div`
  padding: ${getToken("gap", "base")};

  ${getBreakpoint("laptop")} {
    padding: ${getToken("gap", "large")};
  }
`
```
