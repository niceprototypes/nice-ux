# Styles only

```bash
npm install nice-react-styles styled-components
```

```tsx
import { StylesProvider } from "nice-react-styles"

export default function App() {
  return (
    <StylesProvider>
      <Page />
    </StylesProvider>
  )
}
```

```tsx
import styled from "styled-components"
import { getToken } from "nice-react-styles"

const Card = styled.div`
  padding: ${getToken("gap", "base")};
  background: ${getToken("backgroundColor", "alternate")};
  border-radius: ${getToken("borderRadius", "base")};
`
```
