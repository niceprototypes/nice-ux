```tsx
import { setTokens } from "nice-react-styles"

setTokens({
  // STATIC MODULES

  // Basic module
  fontFamily: {
    // Static tokens
    base: "Open Sans",
    code: "Roboto Mono",
    // Custom tokens
    promo: "Billboard Sans Pro",
    numbers: "Calculator Mono",
  },

  // Themed module
  backgroundColor: {
    // Theme objects (static)
    base: {
      // Theme tokens (static)
      day: "#fff",
      night: "#000",
    },
  },

  // Responsive module
  fontSize: {
    // Responsive object (static)
    base: {
      // Responsive tokens (static) — phone is the required base
      phone: "16px",
      laptop: "24px",
      desktop: "32px",
    },
  },

  // CUSTOM MODULES

  brandColor: {
    // Custom keys
    primary: {
      // Theme tokens (static)
      day: "#dc0000",
      night: "#ff5050",
    },
  },
} as const)
```
