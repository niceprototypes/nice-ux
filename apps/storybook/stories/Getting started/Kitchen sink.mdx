## Easy

The minimum: wrap the app in `StylesProvider` and read a token with `getToken`.
Semantic tokens (no variant suffix) flip day/night on their own.

```tsx
// App.tsx

import { StylesProvider, getToken } from "nice-react-styles"
import styled from "styled-components"
import Typography from "nice-react-typography"

const Card = styled.div`
  padding: ${getToken("gap", "large")};
  background-color: ${getToken("backgroundColor")};
  color: ${getToken("color")};
  border-radius: ${getToken("borderRadius", "base")};
`

export default function App() {
  return (
    <StylesProvider>
      <Card>
        <Typography as="h1">Ready to ship</Typography>
      </Card>
    </StylesProvider>
  )
}
```

<Callout type="hint">`StylesProvider` supplies the default token system to every nice component — no `setTokens` needed until you want to override a default.</Callout>

## Medium

Override defaults with `setTokens`, load a font family through the provider, and
go responsive. A `phone` key makes a value responsive; `getBreakpoint` emits the
matching `@media` literal, and a `getToken` inside it resolves to that
breakpoint's variant automatically.

```tsx
// App.tsx

import {
  StylesProvider,
  setTokens,
  getToken,
  getBreakpoint,
} from "nice-react-styles"
import styled from "styled-components"
import Typography from "nice-react-typography"

setTokens({
  fontFamily: {
    base: "Inter", // Default override
  },
  breakpoints: {
    tablet: 700,
    laptop: 1100,
  },
  fontSize: {
    base: { phone: "16px", laptop: "18px" },
    larger: { phone: "32px", laptop: "44px" },
  },
} as const)

const Card = styled.div`
  padding: ${getToken("gap", "large")};
  font-family: ${getToken("fontFamily", "base")};

  ${getBreakpoint("laptop+")} {
    padding: ${getToken("gap", "larger")};
  }
`

export default function App() {
  return (
    <StylesProvider googleFonts="https://fonts.googleapis.com/css2?family=Inter:wght@400;600&display=swap">
      <Card>
        <Typography as="h1" size="larger">Responsive & branded</Typography>
      </Card>
    </StylesProvider>
  )
}
```

<Callout type="learning">A `getToken` call placed inside a `getBreakpoint` block resolves to that breakpoint's variant automatically — your JS and CSS stay on one threshold table.</Callout>

## Hard

Everything at once: static, responsive, themed, and custom tokens; multiple
font families; and a subtree pinned to `night` regardless of the OS preference.
Detection is by shape — a `phone` key makes a value responsive, a `day` key
makes it themed.

```tsx
// App.tsx

import {
  StylesProvider,
  setTokens,
  getToken,
  getBreakpoint,
  Theme,
} from "nice-react-styles"
import styled from "styled-components"
import Typography from "nice-react-typography"
import Icon from "nice-react-icon"

setTokens({
  // Fonts
  fontFamily: {
    base: "Inter",              // Default override
    code: "Roboto Mono",        // Default override
    promo: "Playfair Display",  // Custom token
  },

  // Breakpoints
  breakpoints: {
    tablet: 700,
    laptop: 1100,
    desktop: 1800,
  },

  // Responsive
  fontSize: {
    base: { phone: "16px", laptop: "18px", desktop: "20px" },
    larger: { phone: "32px", laptop: "44px", desktop: "56px" },
  },

  // Themes
  backgroundColor: {
    base: { day: "#ffffff", night: "#0b0b0c" },
  },
  color: {
    base: { day: "#0b0b0c", night: "#f4f4f5" },
  },

  // Custom module
  brandColor: {
    primary: { day: "#dc0000", night: "#ff5050" },
    onBrand: "#ffffff", // static within the same custom module
  },
} as const)

// Semantic tokens (no theme suffix) flip automatically; the responsive
// fontSize resolves per breakpoint via getBreakpoint.
const Card = styled.div`
  padding: ${getToken("gap", "large")};
  background-color: ${getToken("backgroundColor")};
  color: ${getToken("color")};
  border-radius: ${getToken("borderRadius", "base")};
  font-family: ${getToken("fontFamily", "base")};

  ${getBreakpoint("laptop+")} {
    padding: ${getToken("gap", "larger")};
  }
`

export default function App() {
  return (
    <StylesProvider googleFonts="https://fonts.googleapis.com/css2?family=Inter:wght@400;600&family=Playfair+Display:wght@700&family=Roboto+Mono&display=swap">
      <Card>
        <Typography as="h1" size="larger" style={{ fontFamily: getToken("fontFamily", "promo") }}>
          Ready to ship
        </Typography>

        <Typography style={{ color: getToken("brandColor", "primary") }}>
          <Icon name="check" color="success" /> Tokens, fonts, breakpoints, and themes — wired.
        </Typography>

        {/* Pin a subtree to night regardless of the OS preference. */}
        <Theme name="night">
          <Card>
            <Typography>Always dark — pinned with Theme.</Typography>
          </Card>
        </Theme>
      </Card>
    </StylesProvider>
  )
}
```

<Callout type="hint">Semantic tokens like `getToken("color")` already flip day/night on their own. Reach for `Theme`, a component's `theme` prop, or `data-theme` only when you need to *pin* a region against the OS preference.</Callout>
