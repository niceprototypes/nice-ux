# Typography

```bash
npm install nice-react-styles nice-react-typography
```

```tsx
import { StylesProvider } from "nice-react-styles"
import Typography from "nice-react-typography"

<StylesProvider loadFonts>
  <Typography as="h1" size="larger" weight="bold">
    Hello
  </Typography>
  <Typography color="medium">
    Body copy.
  </Typography>
</StylesProvider>
```
