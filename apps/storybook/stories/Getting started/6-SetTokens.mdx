[//]: # (To make your own tokens viewport-aware, pass a breakpoint object &#40;`{ phone, laptop, … }`&#41; as a variant value to `setTokens`. It emits a phone-first default plus the per-breakpoint reassignment cascade — the same shape `tokens.css` ships for the built-in tokens. Breakpoint, theme, and static values can be mixed in one call. `setTokens` injects the generated CSS as a side effect; call it once at module load:)

```bash
npm install nice-react-styles
```

```tsx
import { setTokens } from "nice-react-styles"

setTokens({
  // STATIC MODULES

  // Basic module
  fontFamily: {
    // Static tokens
    base: "Open Sans",
    code: "Roboto Mono",
    // Custom tokens
    promo: "Billboard Sans Pro",
    numbers: "Calculator Mono",
  },

  // Themed module
  backgroundColor: {
    // Theme objects (static)
    base: {
      // Theme tokens (static)
      day: "#fff",
      night: "#000",
    },
  },

  // Responsive module
  fontSize: {
    // Responsive object (static)
    base: {
      // Responsive tokens (static)
      tablet: "16px",
      laptop: "24px",
      desktop: "32px",
    },
  },

  // CUSTOM MODULES

  brandColor: {
    // Custom keys
    primary: {
      // Theme tokens (static)
      day: "#dc0000",
      night: "#ff5050",
    },
  },
} as const)
```
