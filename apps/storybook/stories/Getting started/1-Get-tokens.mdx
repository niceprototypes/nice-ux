## CSS only

### Direct

Import the stylesheet and reference the tokens:

<Install package="styles" showTitle={false} />

```css
@import "nice-styles/tokens.css";

.message {
  color: var(--np--color--base);
}

.message.success {
  color: var(--np--color--success);
}

.message.error {
  color: var(--np--color--error);
}
```

### Local aliases

You may choose to alias the Nice UI tokens you need, and use your aliases everywhere else:

<Install package="styles" showTitle={false} />

```css
@import "nice-styles/tokens.css";

:root {
  --color--base:    var(--np--color--base);
  --color--success: var(--np--color--success);
  --color--error:   var(--np--color--error);
}

.message {
  color: var(--color--base);
}

.message.success {
  color: var(--color--success);
}

.message.error {
  color: var(--color--error);
}
```

## JavaScript

### JS only

Use the `getToken` function and CSS variables provided by `nice-styles` for vanilla JS:

<Install package="styles" showTitle={false} />

```js
import { getToken } from "nice-styles"
import "nice-styles/tokens.css"

// Create HTML element
const card = document.createElement("div")

// Apply token values
card.style.padding = getToken("gap", "large")
card.style.backgroundColor = getToken("backgroundColor")
card.style.borderRadius = getToken("borderRadius", "small")

// Append element to DOM
document.body.append(card)
```

### React only

Create and style a React component with tokens:

<Install package="react-styles" showTitle={false} />

```tsx
import { getToken } from "nice-react-styles"

export default function Card({ children }) {
  return (
    <div
      style={{
        padding: getToken("gap", "large"),
        backgroundColor: getToken("backgroundColor"),
        borderRadius: getToken("borderRadius", "small"),
      }}
    >
      {children}
    </div>
  )
}
```

Then, render the React component inside `StylesProvider`:

```tsx
import { StylesProvider } from "nice-react-styles"

export default function App() {
  return (
    <StylesProvider>
      <Card>…</Card>
    </StylesProvider>
  )
}
```

### React + Styled Components

Create and style a Styled Component with tokens:

<Install package="react-styles styled-components" showTitle={false} />

```tsx
// Card.tsx

import { getToken } from "nice-react-styles"
import styled from "styled-components"

const Card = styled.div`
  padding: ${getToken("gap", "large")};
  background-color: ${getToken("backgroundColor")};
  border-radius: ${getToken("borderRadius", "small")};
`

export default Card
```

Render the Styled Component inside `StylesProvider`:

```tsx
import { StylesProvider } from "nice-react-styles"
import styled from "styled-components"
import Card from "./Card"

export default function App() {
  return (
    <StylesProvider>
      <Page>
        <Card>…</Card>
      </Page>
    </StylesProvider>
  )
}
```
