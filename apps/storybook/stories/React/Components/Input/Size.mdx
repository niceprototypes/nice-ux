import { Meta } from "@storybook/addon-docs/blocks"
import { StoryHeader } from "../../../../src/components/Story"
import Input from "nice-react-input"
import Ink from "../../../../src/nice/Ink"
import { DEMO_CODE_COLUMNS } from "../../../../src/components/DataTable"
import { DataTable } from "../../../../src/components/DataTable"

export const sizes = ["smaller", "small", "base", "large", "larger"]

export const sizeLabel = (s) => s.charAt(0).toUpperCase() + s.slice(1)

export const sizeCode = (s) =>
  s === "base"
    ? `<Input\n  placeholder="Base"\n  onChange={handler}\n/>`
    : `<Input\n  size="${s}"\n  placeholder="${sizeLabel(s)}"\n  onChange={handler}\n/>`

<Meta title="Components/Input/size" />

<StoryHeader title="size" />

<Ink color="light">
  Scales the input's height, padding, and font size using size tokens.
</Ink>

<DataTable
  showHeader={false}
  columns={DEMO_CODE_COLUMNS}
  tables={[{ rows: sizes.map((size) => ({
    id: size,
    values: [
      <Input
        size={size}
        placeholder={sizeLabel(size)}
        onChange={() => {}}
      />,
      sizeCode(size),
    ],
  })) }]}
/>
