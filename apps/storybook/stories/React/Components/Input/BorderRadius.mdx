import { Meta } from "@storybook/addon-docs/blocks"
import Input from "nice-react-input"
import StoryDescription from "../../../../src/components/StoryDescription"
import { DocsRow } from "../../../../src/components/DocsRow"

export const radii = ["small", "base", "large"]

export const radiusLabel = (r) => r.charAt(0).toUpperCase() + r.slice(1)

export const radiusCode = (r) =>
  r === "base"
    ? `<Input\n  placeholder="Base"\n  onChange={handler}\n/>`
    : `<Input\n  borderRadius="${r}"\n  placeholder="${radiusLabel(r)}"\n  onChange={handler}\n/>`

<Meta title="React/Components/Input/borderRadius" />

# Input — BorderRadius

<StoryDescription property="border-radius">
  Controls the corner rounding of the input.
</StoryDescription>

<div>
  {radii.map((radius) => (
    <DocsRow
      previewCell={
        <Input
          borderRadius={radius}
          placeholder={radiusLabel(radius)}
          onChange={() => {}}
        />
      }
      language="tsx"
      code={radiusCode(radius)}
    />
  ))}
</div>
