import { Meta } from "@storybook/addon-docs/blocks"
import { StoryHeader } from "../../../../src/components/Story"
import Input from "nice-react-input"
import Ink from "../../../../src/nice/Ink"
import { DEMO_CODE_COLUMNS } from "../../../../src/components/DataTable"
import { DataTable } from "../../../../src/components/DataTable"

export const statuses = ["base", "disabled", "error", "success", "warning"]

export const statusLabel = (s) => s.charAt(0).toUpperCase() + s.slice(1)

export const statusCode = (s) =>
  s === "base"
    ? `<Input\n  placeholder="Base"\n  onChange={handler}\n/>`
    : `<Input\n  status="${s}"\n  placeholder="${statusLabel(s)}"\n  onChange={handler}\n/>`

<Meta title="Components/Forms/Input/status" />

<StoryHeader title="status" />

<Ink color="light">
  Applies a semantic status style to the input (<code>disabled</code>,{" "}
  <code>error</code>, <code>success</code>, or <code>warning</code>). The
  <code>disabled</code> status also disables the underlying input element.
</Ink>

<DataTable
  showHeader={false}
  columns={DEMO_CODE_COLUMNS}
  tables={[{ rows: statuses.map((status) => ({
    id: status,
    values: [
      <Input
        placeholder={statusLabel(status)}
        status={status === "base" ? undefined : status}
        onChange={() => {}}
      />,
      statusCode(status),
    ],
  })) }]}
/>
