import { Meta } from "@storybook/addon-docs/blocks"
import { useState } from "react"
import { StoryHeader } from "../../../../src/components/Story"
import Input from "nice-react-input"
import Button from "nice-react-button"
import Icon from "nice-react-icon"
import Ink from "../../../../src/nice/Ink"
import { DataTableRow } from "../../../../src/components/DataTable"
import { DEMO_CODE_COLUMNS } from "../../../../src/components/DataTable"

export const ClearableInput = () => {
  const [value, setValue] = useState("Amsterdam")
  return (
    <Input
      aria-label="City"
      value={value}
      onChange={setValue}
      end={
        value ? (
          <Button
            iconLeft="x"
            size="smaller"
            aria-label="Clear city"
            onClick={() => setValue("")}
          />
        ) : null
      }
    />
  )
}

<Meta title="Components/Forms/Input/end" />

<StoryHeader title="end" />

<Ink color="light">
  Content rendered inside the field chrome after the native control, on the
  inline-end side (right in LTR, left in RTL). Same accessibility rule as{" "}
  <code>start</code>: decorative content is hidden by the consumer, interactive
  content carries its own accessible name.
</Ink>

## Interactive: clear button

<Ink color="light">
  A button in the slot stays in the tab order after the native control. Give it
  an <code>aria-label</code> that names what it clears; an icon alone has no
  accessible name.
</Ink>
