import { Meta } from "@storybook/addon-docs/blocks"
import { StoryHeader } from "../../../../src/components/Story"
import Image from "nice-react-image"
import Ink from "../../../../src/nice/Ink"
import { DataTableRow } from "../../../../src/components/DataTable"
import { DEMO_CODE_COLUMNS } from "../../../../src/components/DataTable"
import { toSource } from "../../../../src/services"

export const imageProps = {
  vendor: true,
  alt: "Custom picture element",
}

export const renderImageExpr = `(src, alt) => (
  <picture>
    <source srcSet={src} type="image/jpeg" />
    <img src={src} alt={alt} />
  </picture>
)`

<Meta title="Components/Image/renderImage" />

<StoryHeader title="renderImage" />

<Ink color="light">
  Escape hatch: replace the default <code>&lt;img&gt;</code> /{" "}
  <code>&lt;div&gt;</code> rendering with a custom element. Useful for
  delegating to a framework-specific image component (e.g.{" "}
  <code>next/image</code>, a CDN loader, or a responsive{" "}
  <code>&lt;picture&gt;</code> element). Width, height, backgroundSize,
  borderRadius, and other style props are ignored when <code>renderImage</code>{" "}
  is provided.
</Ink>

<DataTableRow
  columns={DEMO_CODE_COLUMNS}
  values={[
          <Image
      {...imageProps}
      renderImage={(src, alt) => (
        <picture>
          <source
            srcSet={src}
            type="image/jpeg"
          />
          <img
            src={src}
            alt={alt}
            width={240}
            style={{ border: "2px dashed currentColor", padding: 4 }}
          />
        </picture>
      )}
    />,
          toSource("Image", imageProps, {
      raw: { renderImage: renderImageExpr },
    }),
        ]}
/>
