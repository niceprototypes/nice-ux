{/* Columns mirror the module tables above one-for-one so the two line up, which
    is also why columnWidths is their constant rather than a local copy. The CSS
    column is deliberately empty: getHSLA returns a static hsla() string, not a
    var(), so there is no CSS-variable form of these calls to show. */}
<DataTable
  inline
  columns={[
    { key: "swatch", header: null, render: "node" },
    { key: "label", header: "Description", render: "text", size: "small" },
    { key: "css", header: "CSS", render: "node" },
    { key: "call", header: "JS", render: "shiki", language: "tsx" },
    { key: "result", header: "Result", render: "text", code: true, size: "small", color: "lighter", clickToCopy: true },
  ]}
  columnWidths={TOKEN_COLUMN_WIDTHS}
  tables={[{ rows: hslaExamples.map(({ label, call, args }) => {
    const result = getHSLA(...args)
    return {
      id: label,
      values: [
        <PreviewCell
          backgroundColor={result}
          height={getToken("size", "large")}
          borderRadius={getToken("borderRadius", "small")}
        />,
        label,
        null,
        call,
        result,
      ],
    }
  }) }]}
/>
